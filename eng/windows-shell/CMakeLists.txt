cmake_minimum_required(VERSION 3.21)
project(BubuZipShell LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(BubuZipShell SHARED ExplorerCommand.cpp BubuZipShell.def)
target_compile_options(BubuZipShell PRIVATE /W4 /WX /utf-8 /EHsc)
target_compile_definitions(BubuZipShell PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(BubuZipShell PRIVATE shell32 shlwapi ole32)
target_include_directories(BubuZipShell PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
# Keep archive filtering in sync with the application's supported formats.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../src/BubuZip/IShellIntegrationService.cs" FORMATS_SOURCE)
string(REGEX MATCHALL "\"\\.[a-z0-9]+\"" FORMATS "${FORMATS_SOURCE}")
list(TRANSFORM FORMATS PREPEND "L")
list(JOIN FORMATS ", " FORMATS)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ArchiveFormats.h" "#pragma once\ninline constexpr const wchar_t* ArchiveExtensions[] = { ${FORMATS} };\n")
include(CTest)
if(BUILD_TESTING)
  add_executable(ShellTests ShellTests.cpp)
  target_compile_options(ShellTests PRIVATE /W4 /WX /utf-8 /EHsc)
  target_compile_definitions(ShellTests PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_include_directories(ShellTests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
  target_link_libraries(ShellTests PRIVATE shell32 shlwapi ole32)
  add_test(NAME ShellTests COMMAND ShellTests)
endif()
